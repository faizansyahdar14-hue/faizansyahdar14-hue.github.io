const display = document.getElementById('display');

function appendToDisplay(input) {
    if (display.value === '0') {
        display.value = input;
    } else {
        display.value += input;
    }
}

function clearDisplay() {
    display.value = '0';
}

function deleteChar() {
    if (display.value.length === 1) {
        display.value = '0';
    } else {
        display.value = display.value.slice(0, -1);
    }
}

function calculate() {
    try {
        // eval() menghitung string matematika
        display.value = eval(display.value);
    } catch (error) {
        display.value = "Error";
        setTimeout(clearDisplay, 1500); // Reset otomatis setelah 1.5 detik jika error
    }
}
